async function submitToCoach(){
  // 入力値の取得
  state.log.distance = Number(document.getElementById("dist").value) || 0;
  state.log.zone = document.getElementById("zone").value;
  state.log.note = document.getElementById("note").value;
  save();

  if(!state.apiKey){
    alert("「SETTINGS / CONNECT」タブで OpenAI API Key を設定してください。");
    go("connect");
    return;
  }

  state.loading = true;
  state.coachResponse = "ChatGPT 監督が本日の練習結果を厳密に分析中...";
  render();

  const prompt = `あなたは陸上長距離・中距離ランナーの「超厳しめ・忖度なしのAIヘッドコーチ」です。
ユーザー（ランナー）から本日の練習報告が届きました。距離の長さだけでなく、目的達成度、疲労、痛み、睡眠から総合的に評価し、短く的確（200文字以内）に厳しい愛のあるフィードバックを行ってください。

【本日の報告データ】
- 練習距離: ${state.log.distance} km
- 目的/ゾーン: ${state.log.zone}
- 主観的きつさ (RPE 1-10): ${state.rpe}
- 睡眠時間: ${state.sleep} 時間
- 疲労感 (1-5): ${state.fatigue}
- 痛み・違和感 (0-3): ${state.pain}
- ひとこと・メモ: ${state.log.note || "なし"}

アドバイスの方向性:
- 痛みが1以上の場合は怪我予防を最優先で警告すること。
- RPEや疲労度が高い場合、翌日のリカバリーを指示すること。
- 距離稼ぎ目的の無理な走りは一蹴すること。`;

  try {
    const res = await fetch("https://api.openai.com/v1/chat/completions", {
      method: "POST",
      headers: {
        "Content-Type": "application/json",
        "Authorization": `Bearer ${state.apiKey}`
      },
      body: JSON.stringify({
        model: state.model || "gpt-4o-mini",
        messages: [{ role: "user", content: prompt }],
        temperature: 0.7
      })
    });

    const data = await res.json();
    if (data.error) {
      state.coachResponse = `エラーが発生しました:\n${data.error.message}`;
    } else {
      state.coachResponse = data.choices[0].message.content;
      toast("監督からフィードバックが届きました！");
    }
  } catch (err) {
    state.coachResponse = `通信エラーが発生しました: ${err.message}`;
  } finally {
    state.loading = false;
    save();
    render();
  }
}

function render(){
  nav();
  document.getElementById("main").innerHTML={home,plan,log,growth,connect,zones,roadmap}[state.tab]();
}

render();
